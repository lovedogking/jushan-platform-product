## 一、SSH 登录服务器

```bash
# 密码登录

# 密钥登录（脚本用）
```

## 二、后端

```bash
# 编译
mvn clean package -pl parking-boot -am -Dmaven.test.skip=true

# 只检查编译
mvn compile -pl parking-system

# 上传+部署
scp -i docs/服务器/密钥对.pem parking-boot/target/parking-boot-1.0.0-SNAPSHOT.jar root@120.26.3.4:/opt/jushan/parking-boot/target/
ssh -i docs/服务器/密钥对.pem root@120.26.3.4 "cd /opt/jushan && docker compose -f docker-compose.prod.yml --env-file .env.production up -d --build backend"

# 仅重启
ssh -i docs/服务器/密钥对.pem root@120.26.3.4 "docker restart jushan-backend"

# 查看日志
ssh -i docs/服务器/密钥对.pem root@120.26.3.4 "docker logs --tail 50 jushan-backend"
ssh -i docs/服务器/密钥对.pem root@120.26.3.4 "docker logs -f jushan-backend"
```

## 六、Docker


## 七、健康检查

```bash
# 后端（外部——会被 Nginx 拒绝，预期返回 403）
curl http://120.26.3.4/api/actuator/health
# → 403 Forbidden（Nginx 仅允许内网访问 Actuator）

# 后端（SSH 进服务器执行——正确方式）
ssh -i docs/服务器/密钥对.pem root@120.26.3.4 "curl -s http://127.0.0.1:8080/actuator/health"

# 前端
curl http://120.26.3.4/

# DA 进程存活确认
ssh -i docs/服务器/密钥对.pem root@120.26.3.4 "ps aux | grep device-access | grep -v grep"

# DA 端口确认
ssh -i docs/服务器/密钥对.pem root@120.26.3.4 "curl -s -o /dev/null -w '%{http_code}' http://127.0.0.1:8082/actuator/health"
# → 当前返回 404（DA 未配置 health endpoint）

# Docker 容器健康状态
ssh -i docs/服务器/密钥对.pem root@120.26.3.4 "docker ps --format 'table {{.Names}}\t{{.Status}}'"
```

```bash

ssh -i docs/服务器/密钥对.pem root@120.26.3.4 "cd /opt/jushan && docker compose -f docker-compose.prod.yml --env-file .env.production up -d --build backend"

# 前端

rsync -avz --delete -e "ssh -i ../docs/服务器/密钥对.pem" dist/ root@120.26.3.4:/opt/jushan/frontend/dist/ && \
cd ..
```
